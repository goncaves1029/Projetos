-- 1. Criar o banco de dados
CREATE DATABASE IF NOT EXISTS igreja
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE igreja;

-- 2. Criar a tabela presencas
CREATE TABLE IF NOT EXISTS presencas (
  id INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  culto VARCHAR(100) DEFAULT NULL,
  data DATE DEFAULT NULL,
  hora TIME DEFAULT NULL,
  adultos INT DEFAULT 0,
  criancas_templo INT DEFAULT 0,
  kids_baby INT DEFAULT 0,
  kids INT DEFAULT 0,
  link INT DEFAULT 0,
  louvor INT DEFAULT 0,
  comunicacao INT DEFAULT 0,
  voluntarios INT DEFAULT 0,
  pastor INT DEFAULT 0,
  pregador TEXT DEFAULT NULL,
  carros INT DEFAULT 0,
  motos INT DEFAULT 0,
  bicicletas INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
